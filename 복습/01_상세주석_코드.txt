# 파일과 폴더의 위치를 다루는 도구를 꺼냅니다.
import os
# 많은 숫자를 함께 계산하는 NumPy를 np라는 이름으로 씁니다.
import numpy as np
# 엑셀처럼 생긴 표를 다루는 pandas를 pd라는 짧은 이름으로 씁니다.
import pandas as pd

# 현재 코드 파일의 폴더에서 한 단계 위로 올라가 데이터 폴더 위치를 만듭니다.
# __file__: 이 코드 파일 / abspath: 전체 경로 / dirname: 폴더 / join: 경로 연결.
DATA = os.path.join(
    os.path.dirname(os.path.abspath(__file__)), "..", "수업용데이터"
)
# 00에서 손질한 clean이 아니라 원본 센서 CSV를 이 프로그램에서 새로 읽습니다.
df = pd.read_csv(os.path.join(DATA, "11_설비센서_ai4i.csv"), encoding="utf-8-sig")

# 입력 공기온도를 배열로 꺼내 실수형으로 바꿉니다. 입력 x와 실제 정답 y는 다릅니다.
x = df["공기온도"].values.astype(
    float
)
# 실제 공정온도를 정답 배열 y에 담습니다. 이 값 자체는 학습으로 수정하지 않습니다.
y = df["공정온도"].values.astype(float)
# 개수와 가장 작은 값·큰 값을 보여줘 데이터의 대략적인 범위를 확인합니다.
print(
    "설비",
    len(x),
    "대 / 공기온도",
    x.min(),
    "~",
    x.max(),
    "/ 공정온도",
    y.min(),
    "~",
    y.max(),
)
print("앞 5대  공기온도:", x[:5])
print("앞 5대  공정온도:", y[:5])

# x는 재료, w는 곱하는 손잡이, b는 더하는 손잡이입니다.
# def는 계산법을 이름으로 묶고 return은 계산한 답을 돌려줍니다.
def 예측(x, w, b):
    # x가 배열이면 같은 계산을 각 숫자마다 해줍니다. 예측값을 반환합니다.
    return w * x + b

print("\n[1] 일단 감으로 w=1, b=10 넣어보기 ('공정온도 = 공기온도 + 10')")
# 아직 학습하지 않고 w=1, b=10으로 계산한 예측의 앞 5개를 봅니다.
print("예측 (앞 5대):", 예측(x, 1.0, 10.0)[:5])
# 앞 5개 실제 정답과 비교합니다. 비교할 때 위치가 서로 맞아야 합니다.
print("실제 (앞 5대):", y[:5])

# 예측과 실제 y를 비교해 벌점 하나를 계산하는 함수입니다.
def 손실(x, y, w, b):
    # 같은 위치의 실제 정답에서 예측을 뺍니다. 양수면 실제보다 낮게 예측한 것입니다.
    틀린정도 = y - 예측(x, w, b)
    # **2는 제곱, mean은 평균입니다. 각 오차를 제곱한 뒤 평균을 냅니다.
    return np.mean(틀린정도**2)

print("\n[2] 여러 직선의 손실 점수")
# (w, b) 후보 쌍을 하나씩 넣어 어느 직선의 벌점이 작은지 확인합니다.
for w, b in [(1.0, 10.0), (1.0, 9.0), (1.0, 11.0), (0.0, 310.0)]:
    print(
        f"    w={w}, b={b:<5} → 손실 {손실(x, y, w, b):.4f}"
    )

print("\n[3] b 를 살짝 밟아보기 (w=1.0 고정, 지금 b=9.0 이라고 치고)")
print(f"    b=8.9 → 손실 {손실(x, y, 1.0, 8.9):.4f}")
print(f"    b=9.0 → 손실 {손실(x, y, 1.0, 9.0):.4f}   ← 지금 자리")
print(f"    b=9.1 → 손실 {손실(x, y, 1.0, 9.1):.4f}")
print(f"    b=9.2 → 손실 {손실(x, y, 1.0, 9.2):.4f}")
print(f"    b=10.0 → 손실 {손실(x, y, 1.0, 10.0):.4f}")
print(f"    b=11.0 → 손실 {손실(x, y, 1.0, 11.0):.4f}")

# h는 방향을 살펴볼 작은 거리입니다. 실제 수정 보폭인 lr과 역할이 다릅니다.
h = 0.0001

# w와 b를 각각 혼자 조금 움직여 손실의 변화 방향을 알아봅니다.
def 기울기_밟아보기(x, y, w, b):
    # w만 +h와 -h로 바꿉니다. b는 그대로입니다. 두 위치 사이 거리는 2h입니다.
    gw = (손실(x, y, w + h, b) - 손실(x, y, w - h, b)) / (
        2 * h
    )
    # 이번에는 b만 +h와 -h로 바꾸고 w는 그대로 둡니다.
    gb = (손실(x, y, w, b + h) - 손실(x, y, w, b - h)) / (
        2 * h
    )
    # 기울기 두 값을 정해진 순서로 돌려줍니다.
    return gw, gb

# 현재 w=1, b=9에서 기울기 둘을 구하고 gw와 gb로 받습니다.
gw, gb = 기울기_밟아보기(x, y, 1.0, 9.0)
print(f"\n    w=1.0, b=9.0 에서  b 방향 기울기 = {gb:.3f} (음수 → b 를 키워라)")
print(f"                       w 방향 기울기 = {gw:.3f} (음수 → w 도 키워라)")

# 기울기에 곱해 실제 이동 크기를 조절할 학습률입니다.
lr = 0.00001
# 한 걸음 실험의 시작 손잡이 값을 정합니다.
w, b = 1.0, 9.0
print("\n[4-1] 한 걸음 (lr=0.00001)")
print(f"    이전: w={w:.5f}, b={b:.5f}, 손실 {손실(x, y, w, b):.4f}")
# 학습률에 기울기를 곱한 만큼 반대 방향으로 w를 움직입니다.
w = w - lr * gw
# 같은 방식으로 b도 움직입니다. gw와 gb는 둘 다 수정 전 위치에서 구했습니다.
b = b - lr * gb
print(
    f"    이후: w={w:.5f}, b={b:.5f}, 손실 {손실(x, y, w, b):.4f}   ← 줄었죠? 이걸 반복하면 됩니다"
)

# 앞 실험의 끝에서 이어가지 않고 시작값을 다시 설정합니다.
w, b = 1.0, 9.0
# 일부 숫자 계산 경고만 잠깐 숨깁니다. 잘못된 계산을 고쳐주는 기능은 아닙니다.
with np.errstate(over="ignore", invalid="ignore"):
    # 정해진 횟수만큼 반복합니다. _는 반복 번호 자체는 사용하지 않겠다는 관례입니다.
    for _ in range(20):
        # 여기서는 학습률 0.001로 동시에 두 손잡이를 갱신합니다.
        gw, gb = 기울기_밟아보기(x, y, w, b)
        w, b = w - 0.001 * gw, b - 0.001 * gb
print(
    "\n[5] 보폭을 0.001로 키워 20걸음 -> 손실:",
    손실(x, y, w, b),
    "<- 숫자가 폭발(발산)",
)

# 공기온도 전체의 평균을 구합니다. 새 눈금의 가운데 기준입니다.
m = x.mean()
# 퍼진 정도인 표준편차를 구합니다. 새 눈금의 한 칸 크기입니다.
s = x.std()
# 평균을 빼고 표준편차로 나누어 표준화합니다. y는 바꾸지 않습니다.
z = (x - m) / s
print(f"표준화: 평균 {m:.2f}, 표준편차 {s:.2f}")
print(f"x 앞 3개: {x[:3]} -> z 앞 3개: {z[:3].round(3)}")

# 함수 정의만으로 학습이 시작되지는 않습니다. 아래에서 학습(z, y)를 불러야 실행됩니다.
# lr, epochs, 보여주기는 생략했을 때 사용할 기본 설정입니다.
def 학습(z, y, lr=0.1, epochs=300, 보여주기=True):
    # 가중치와 절편을 0에서 시작합니다. 표준화한 입력에 맞는 손잡이를 배웁니다.
    w, b = 0.0, 0.0
    # range(300)은 0부터 299까지입니다. 전체 사례로 배우고 수정하는 일을 반복합니다.
    for epoch in range(epochs):
        # 함수의 첫 매개변수 이름이 x여도 여기서는 z 배열을 넘기므로 표준화한 값으로 계산합니다.
        gw, gb = 기울기_밟아보기(z, y, w, b)
        # 학습률에 기울기를 곱한 만큼 반대 방향으로 w를 움직입니다.
        w = w - lr * gw
        # 같은 방식으로 b도 움직입니다. gw와 gb는 둘 다 수정 전 위치에서 구했습니다.
        b = b - lr * gb
        # 출력 스위치가 켜져 있고 지정한 반복 번호일 때만 화면에 진행 상황을 보여줍니다.
        if 보여주기 and epoch in (0, 1, 2, 5, 10, 30, 100, 299):
            print(
                f"    epoch {epoch:3d}  w={w:7.3f}  b={b:8.3f}  손실 {손실(z, y, w, b):10.4f}"
            )
    # 학습을 마친 두 손잡이를 돌려줍니다.
    return w, b

print("\n[6] 경사하강 (표준화한 z 로, lr=0.1)")
# 학습 함수를 실행하고 표준화 눈금의 가중치 w_z와 절편 b_z를 받습니다.
w_z, b_z = 학습(z, y)

# 표준화한 x 대신 원래 x를 곱해도 되도록 가중치를 변환합니다.
w_gd = w_z / s
# 입력 눈금에서 평균을 뺐던 효과까지 고려하여 절편도 함께 바꿉니다.
b_gd = b_z - w_z * m / s

# 표준화한 x 대신 원래 x를 곱해도 되도록 가중치를 변환합니다.
w_gd = w_z / s
# 입력 눈금에서 평균을 뺐던 효과까지 고려하여 절편도 함께 바꿉니다.
b_gd = b_z - w_z * m / s
print(f"\n    결과: 공정온도 ≈ {w_gd:.4f} × 공기온도 + {b_gd:.4f}")
print(
    f"    손실: {손실(x, y, w_gd, b_gd):.4f}  (감으로 넣은 (1, 10) 은 0.8941 → 0.8921. 감이 꽤 좋았던 셈이죠)"
)

print("\n[7] 보폭(lr)을 바꿔 300걸음 후 손실")
# 각 학습률마다 새로 300번 배우게 합니다. 이전 학습 결과를 이어 쓰지 않습니다.
for lr in [0.001, 0.01, 0.1, 1.05]:
    w_t, b_t = 학습(z, y, lr=lr, 보여주기=False)
    L = 손실(z, y, w_t, b_t)
    # NaN이거나 손실이 백만보다 크면 이 예제에서는 발산으로 표시합니다.
    if np.isnan(L) or L > 1e6:
        판정 = "발산 (튕겨 나감)"
    # 앞 조건은 아니지만 손실이 1보다 크면 아직 충분히 배우지 못했다고 표시합니다.
    elif L > 1.0:
        판정 = "굼벵이 (300걸음으론 아직 멀었음)"
    else:
        판정 = "좋음 (바닥)"
    print(f"    lr={lr:<6} → 손실 {L:>14.4f}  {판정}")

# 중심에서 벗어난 x와 y가 함께 움직이는 정도를 이용해 최적 직선의 w를 구합니다.
w_공식 = np.sum((x - x.mean()) * (y - y.mean())) / np.sum((x - x.mean()) ** 2)
# 직선이 데이터 평균 지점을 지나도록 b를 구합니다.
b_공식 = y.mean() - w_공식 * x.mean()
print("\n[8] 대조")
print(f"    걸어서   w={w_gd:.4f}  b={b_gd:.4f}")
print(f"    공식으로 w={w_공식:.4f}  b={b_공식:.4f}")
# 두 차이가 모두 0.001보다 작은지 and로 확인합니다.
print(
    "    같은가? ", abs(w_gd - w_공식) < 1e-3 and abs(b_gd - b_공식) < 1e-3
)

print("\n[9] 예측")
# 학습이 아니라 예측입니다. 새 공기온도 세 개를 배운 식에 넣습니다.
for 새온도 in [298.0, 300.0, 303.0]:
    print(f"    공기온도 {새온도} → 예측 공정온도 {예측(새온도, w_gd, b_gd):.2f}")

# 모델의 제곱오차 합을 평균만 답하는 기준과 비교하는 채점 함수입니다.
def R2(x, y, w, b):
    # 모든 사례에서 모델의 제곱오차를 더합니다.
    못맞힌것 = np.sum((y - 예측(x, w, b)) ** 2)
    # 매번 실제 정답의 평균만 답했을 때의 제곱오차를 더합니다.
    평균으로찍기 = np.sum((y - y.mean()) ** 2)
    # 평균 기준과 똑같으면 0, 완벽하게 맞으면 1입니다. 분모가 0이면 별도 처리가 필요합니다.
    return 1 - 못맞힌것 / 평균으로찍기

# 배웠던 200개로 다시 채점합니다. 독립된 시험 점수는 아닙니다.
r2 = R2(x, y, w_gd, b_gd)
print(
    f"    R² = {r2:.4f}  ← 공정온도의 흩어짐 중 약 {r2 * 100:.0f}% 를 공기온도 하나로 설명"
)

print("\n[9-1] 설비 2대만 주고 배우게 하면 R² 는 몇 점일까")
# 입력과 정답에서 같은 앞 두 위치를 함께 꺼내 짝을 유지합니다.
x2, y2 = x[:2], y[:2]
# 두 점의 세로 차이 / 가로 차이로 기울기를 구합니다. 두 입력이 같으면 쓸 수 없습니다.
w2 = (y2[1] - y2[0]) / (x2[1] - x2[0])
# 첫 점을 정확히 통과하도록 절편을 계산합니다.
b2 = y2[0] - w2 * x2[0]
# 방금 만든 두 점 전용 직선으로 두 정답을 예측합니다.
예측2 = 예측(x2, w2, b2)
print(
    f"    배운 2대로 채점: R² = {1 - np.sum((y2 - 예측2) ** 2) / np.sum((y2 - y2.mean()) ** 2):.4f}  ← 만점"
)

print(
    f"    같은 직선으로 200대 전부 채점: R² = {R2(x, y, w2, b2):.4f}  ← 200대로 배운 0.7989 보다 한참 아래"
)
print(
    f"    참고: 2대로 얻은 직선은 w={w2:.3f}, b={b2:.1f} / 200대로 얻은 직선은 w={w_gd:.3f}, b={b_gd:.1f}"
)
print()

# 학습률만 바꿔 각각 처음부터 학습하며 손실을 비교하는 마지막 실험입니다.
for lr in [0.1, 0.5, 0.9, 1.0, 1.05]:
    w_t, b_t = 학습(z, y, lr=lr, epochs=300, 보여주기=False)

    L = 손실(z, y, w_t, b_t)

    print(f"lr={lr} → 손실 {L:.4f}")
