# 파일과 폴더의 위치를 다루는 도구를 꺼냅니다.
import os
# 엑셀처럼 생긴 표를 다루는 pandas를 pd라는 짧은 이름으로 씁니다.
import pandas as pd
# 많은 숫자를 함께 계산하는 NumPy를 np라는 이름으로 씁니다.
import numpy as np

# 현재 코드 파일의 폴더에서 한 단계 위로 올라가 데이터 폴더 위치를 만듭니다.
# __file__: 이 코드 파일 / abspath: 전체 경로 / dirname: 폴더 / join: 경로 연결.
DATA = os.path.join(
    os.path.dirname(os.path.abspath(__file__)), "..", "수업용데이터"
)

# 파일 이름을 주면 데이터 폴더와 연결하여 위치를 돌려주는 작은 함수입니다.
def path(file_name):
    """data/ 폴더 안의 파일 경로를 만들어 주는 작은 도우미 함수"""
    return os.path.join(DATA, file_name)

print("=" * 60)
print("[0] 파이썬 표현 미리 보기")

# =는 이름에 값을 담는 기호입니다. x라는 이름으로 3.14159를 기억합니다.
x = 3.14159
# f 문자열의 {} 안에 값을 넣습니다. .2f는 소수 두 자리, .1%는 백분율 한 자리입니다.
print(
    f"    ① {x:.2f}  ← .2f = 소수 둘째 자리까지 | {x:8.3f} ← 8칸 폭, 셋째 자리 | {7:>4d} ← 정수를 4칸 오른쪽 정렬 | {0.256:.1%} ← 퍼센트"
)

# [10, 20, 30, 40]이라는 목록을 숫자 계산용 배열로 만듭니다.
a = np.array([10, 20, 30, 40])
# a > 20은 자리마다 질문합니다. a[조건]은 True인 자리의 숫자만 고릅니다.
print("    ② a > 20 은", a > 20, "→ a[a > 20] 은", a[a > 20])

# n에 1, 2, 3을 넣고 각각 n*n을 계산하여 [1, 4, 9]를 만듭니다.
square = [n * n for n in [1, 2, 3]]
print("    ③ [n*n for n in [1,2,3]] →", square)

# 작은 괄호는 이름과 숫자를 한 쌍으로 묶습니다. 큰 대괄호는 세 쌍의 목록입니다.
duo = [("온도", 0.5), ("압력", -2.0), ("진동", 1.2)]
# t는 한 쌍, t[1]은 그 쌍의 두 번째 숫자입니다. abs로 크기를 구합니다.
# sorted는 작은 값부터 정렬하므로 -를 붙여 절댓값이 큰 것이 앞에 오게 합니다.
print(
    "    ④ 절댓값 큰 순 정렬:", sorted(duo, key=lambda t: -abs(t[1]))
)

score = 0.83
# score가 0.8보다 크면 좋음, 아니면 보통이라는 글자를 선택합니다.
print("    ⑤", "좋음" if score > 0.8 else "보통")

b = np.array([1, 2, 3, 4, 5])
# [:3]은 앞 3개, [3:]은 3번 위치부터 끝, [::-1]은 역순입니다.
print("    ⑥ b[:3] =", b[:3], "| b[3:] =", b[3:], "| b[::-1] =", b[::-1])

# 대괄호 안의 두 목록이 각각 한 행입니다. 전체는 2행 3열입니다.
M = np.array([[1, 2, 3], [4, 5, 6]])
# 쉼표 앞은 행, 뒤는 열입니다. :만 쓰면 해당 방향 전체를 선택합니다.
print(
    "    ⑦ M[:, 0] =",
    M[:, 0],
    "| M[0, :] =",
    M[0, :],
    "| M.shape =",
    M.shape,
    "(행 2, 열 3)",
)

print(df.describe().round(2))

print("[3] 열 하나 = df['열이름']  (Series 라는 1차원 표)")

print(df["공정온도"].head(3))

print("[3-1] 조건으로 행 고르기 = df[조건]  ( 문법 ② 의 pandas 판)")

    "/ 전체 평균:",

# 글자로 된 열 둘을 제외한 표를 새 이름에 담습니다. df 자체를 지우는 명령은 아닙니다.
number_columns = df.drop(columns=["설비ID", "타입"])

print("[4] 상관계수 표")
# 숫자 열끼리의 상관계수를 표로 계산하고 소수 두 자리로 반올림합니다.
print(number_columns.corr().round(2))

# 실험용 복사본을 만듭니다. 원본 df는 그대로 보관합니다.
fake = df.copy()
# range의 끝은 포함되지 않으므로 +1을 하여 1부터 행 개수까지 만듭니다.
fake["설비번호"] = range(1, len(df) + 1)
# 설비번호에 0을 곱하므로 그 영향은 없습니다. 공기온도*2+10을 베낀 열입니다.
# 복사 관계라 상관은 1이지만 설비번호가 온도를 일으킨 증거는 아닙니다.
fake["번호_상관"] = (
    fake["설비번호"] * 0 + fake["공기온도"] * 2 + 10
)
print()
print("[4-1] 상관은 높은데 원인이 아닌 경우")
# 두 열의 상관계수를 계산합니다. 이 구성에서는 약 1입니다.
print(
    "    공기온도 vs 번호_상관 의 상관계수:",
    round(fake["공기온도"].corr(fake["번호_상관"]), 4),
)

# 센서 예측용 파일과 별개인 전처리 연습 파일을 읽습니다.
dirty = pd.read_csv(path("12_제조센서_전처리.csv"), encoding="utf-8-sig")
print("=" * 60)
print("[5] 지저분한 데이터 원본")
print(dirty)

print()
print("[5-1] 결측(NaN) 개수 — 열마다")
# isna는 빈칸 여부, sum은 True 개수를 셉니다. 열마다 빈칸 수가 나옵니다.
print(
    dirty.isna().sum()
)

print()
print("[5-2] 이상값 — 물리적으로 말이 안 되는 값 찾기")
print("    회전수가 음수인 행:")
# 이 데이터에서는 회전수에 음수가 있으면 잘못된 기록으로 다룹니다.
print(dirty[dirty["회전수"] < 0])
print("    진동이 유난히 큰 행:")
# 진동이 10보다 큰 행을 찾아 보여주기만 합니다. 아직 삭제하거나 바꾸지 않습니다.
print(
    dirty[dirty["진동"] > 10]
)

print()
print("[5-3] 중복 행 — 같은 측정이 두 번 들어간 경우")
# 처음 나온 행은 남기고, 앞에서 나온 것과 똑같은 추가 행의 개수를 셉니다.
print(
    "    중복 행 수:", dirty.duplicated().sum()
)
# keep=False는 중복된 무리의 모든 행을 표시하여 함께 살펴보게 합니다.
print(
    dirty[dirty.duplicated(keep=False)]
)

# 정리할 사본을 만듭니다. dirty의 원래 내용은 건드리지 않습니다.
clean = dirty.copy()
# 완전히 같은 행이 여러 번 있으면 기본적으로 첫 번째만 남깁니다.
clean = clean.drop_duplicates()
# loc[행 조건, 열 이름]으로 고칠 칸을 고릅니다. 음수 회전수만 빈칸 NaN으로 바꿉니다.
clean.loc[clean["회전수"] < 0, "회전수"] = np.nan
# 온도, 압력, 회전수 열을 하나씩 꺼내 같은 빈칸 채우기 작업을 반복합니다.
for 열 in ["온도", "압력", "회전수"]:
    # 비어 있지 않은 값들을 정렬했을 때 가운데 값을 구합니다.
    중앙값 = clean[열].median()
    # 이 열의 빈칸을 방금 구한 중앙값으로 채우고, 결과를 같은 열에 다시 담습니다.
    clean[열] = clean[열].fillna(중앙값)
    print(f"    {열}: 결측을 중앙값 {중앙값} 로 채움")

print()
print("[5-4] 처리 후")
print(clean)
# 첫 sum은 열별 빈칸 수, 두 번째 sum은 그것들을 합친 전체 빈칸 수입니다.
print("    결측 남은 개수:", int(clean.isna().sum().sum()), "/ 행 수:", len(clean))

print("=" * 60)
print("[6] 문제 정의")
# 세 겹 따옴표는 여러 줄의 글을 하나의 문자열로 담습니다. print로 안내문을 출력합니다.
print("""

      입력 X = 공기온도 (나중엔 회전수·토크·공구마모도 추가)

      → 03 파일에서 로지스틱 회귀로 풉니다.

  판단 기준 한 줄: 정답 열이 '연속된 숫자'면 회귀, '몇 가지 중 하나'면 분류.
  (온도·개수·무게·가격 → 회귀 / 고장·양품·등급·합격 → 분류)
""")
